# -*- coding: utf-8 -*-

from Npp import editor
import re

    # REEMPLAZOS DE TEXTO
reemplazos = {
    "00.": "",
    "Lista de reproducción": "",
}

# =========================================================
# 2. EXPRESIONES REGULARES
# =========================================================

regex_reemplazos = [
    (r'^\s*<location>', ''),
    (r'^\s*<title>', ''),
    (r'(?s)</title>.*?<track>', ''),
    (r'<[^>]+>', ''),
    (r'^[ \t]+', ''),
    (r'Lista\s+de\s+reproducción', ''),
]




# ==========================================
# EJECUTAR EXPRESIONES REGULARES
# ==========================================

for buscar, reemplazar in regex_reemplazos:
    editor.rereplace(buscar, reemplazar)
    
# ==========================================
# EJECUTAR REEMPLAZOS SIMPLES
# ==========================================

if reemplazos:
    patron = r'|'.join(re.escape(x) for x in reemplazos.keys())

    def reemplazar(match):
        return reemplazos[match.group(0)]

    editor.rereplace(patron, reemplazar)
